async function pilihDanTampilkanLp2bPoktan(lat, lon, dataPoktan) { showStatusModal("Memuat Analisis...", "Mengambil poligon LP2B untuk titik ini...", true); const lp2b = await muatDataLP2BGlobal(); document.getElementById('statusModal').style.display = 'none'; // Hapus poligon sebelumnya agar ketika klik pin lain, poligon lama langsung hilang listLp2bLayer.clearLayers(); const pt = turf.point([lon, lat]); let targetPolygonFeature = null; let masukLp2b = false; if (lp2b && lp2b.features) { // 1. Cari poligon yang tepat berada di dalam area titik tersebut for (let f of lp2b.features) { try { if (turf.booleanPointInPolygon(pt, f)) { masukLp2b = true; targetPolygonFeature = f; break; } } catch (err) {} } // 2. Jika titik berada di luar poligon persis, ambil poligon terdekatnya if (!targetPolygonFeature) { let minDistance = Infinity; lp2b.features.forEach(f => { try { const dist = turf.pointToPolygonDistance(pt, f, { units: 'kilometers' }); if (dist < minDistance) { minDistance = dist; targetPolygonFeature = f; } } catch (err) {} }); } } // TAMPILKAN HANYA 1 POLIGON YANG KENA PIN TERSEBUT (Tanpa poligon sekitar) if (targetPolygonFeature) { L.geoJSON(targetPolygonFeature, { style: { color: masukLp2b ? "#27ae60" : "#e67e22", weight: 4, fillColor: masukLp2b ? "#27ae60" : "#e67e22", fillOpacity: 0.6 } }).addTo(listLp2bLayer); } mapList.setView([lat, lon], 16); }
